```julia
function add_optimality_cut!(master::BendersMaster, epigraph::Symbol, cost_k::Real,
                              grad_k::AbstractVector{<:Real}, z_k::AbstractVector{<:Real})
    epigraph in (:op, :x) || throw(ArgumentError("..."))
    length(grad_k) == master.T || throw(ArgumentError("..."))
    length(z_k) == master.T || throw(ArgumentError("..."))
    isfinite(cost_k) || throw(ArgumentError("..."))
    all(isfinite, grad_k) || throw(ArgumentError("..."))
    all(isfinite, z_k) || throw(ArgumentError("..."))
    α = epigraph === :op ? master.α_op : master.α_x
    @constraint(master.model,
        α >= cost_k + sum(grad_k[t] * (master.z[t] - z_k[t]) for t in 1:master.T))
    push!(master.cuts, (; kind = :optimality, epigraph, cost_k,
                         grad_k = Vector{Float64}(grad_k), z_k = Vector{Float64}(z_k)))
    return master
end

function add_feasibility_cut!(master::BendersMaster, v_k::Real, u_k::AbstractVector{<:Real},
                               z_k::AbstractVector{<:Real})
    length(u_k) == master.T || throw(ArgumentError("..."))
    length(z_k) == master.T || throw(ArgumentError("..."))
    isfinite(v_k) || throw(ArgumentError("..."))
    all(isfinite, u_k) || throw(ArgumentError("..."))
    all(isfinite, z_k) || throw(ArgumentError("..."))
    @constraint(master.model,
        v_k + sum(u_k[t] * (master.z[t] - z_k[t]) for t in 1:master.T) <= 0)
    push!(master.cuts, (; kind = :feasibility, v_k, u_k = Vector{Float64}(u_k),
                         z_k = Vector{Float64}(z_k)))
    return master
end
```
