FollowerLP{Z,C} fields: model::Model, x_inv::VariableRef, x_op::Vector{VariableRef},
z::Z (Parameter-typed), coupling::C, T::Int, corridor_cap::Float64, x_inv_max::Float64.

build_follower(; T, corridor_cap, x_inv_max, c_inv, c_op) -> FollowerLP:
  boundary guards -> Model(select_optimizer(LP())) [INFRA-02] -> @variable 0<=x_inv<=x_inv_max,
  x_op[t]>=0, z[t] in Parameter(0.0) -> @constraint invest_op[t]: x_op[t]<=corridor_cap*x_inv,
  coupling[t]: x_op[t]==z[t] -> @objective Min c_inv*x_inv + sum(c_op[t]*x_op[t]).

solve_follower!(f, z_trial) -> NamedTuple: length guard -> set_parameter_value.(f.z, z_trial)
  -> optimize!(f.model) [DIRECT call, NEVER solve_with_retry! — infeasible branch must be
  OBSERVED, not retried away] -> three-way branch:
    feasible (is_solved_and_feasible(f.model; dual=true)): (; feasible=true,
      cost=objective_value(f.model), π_s=dual.(f.coupling))
    infeasible (dual_status(f.model)==MOI.INFEASIBILITY_CERTIFICATE): v=dual_objective_value(f.model),
      u=dual.(f.coupling); ENFORCE isfinite(v) && v>0 && all(isfinite,u) or loud error;
      return (; feasible=false, v, u)
    else: loud error naming termination_status/primal_status/dual_status/raw_status.
