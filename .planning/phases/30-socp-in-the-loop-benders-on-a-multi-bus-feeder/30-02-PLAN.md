explicit bound validated) pays the extra solve. This avoids retrofitting/rewriting dozens of

model's own declared lower bound) is NOT part of this plan — it is universal and
`bounds_ctx`-independent, and lives in `benders.jl` (plan 30-04) since it reads per-iteration
Benders quantities this file has no access to.

Current `src/planning/master.jl` (full file already read — see `@src/planning/master.jl`):
```julia
function build_master(; T::Int, c_y::Real, y_max::Real, α_op_lb::Real, α_x_lb::Real)
    T >= 1 || throw(ArgumentError(...))
    y_max > 0 || throw(ArgumentError(...))
    c_y >= 0 || throw(ArgumentError(...))
    model = Model(select_optimizer(LP()))
    @variable(model, 0 <= y_inv <= y_max)
    @variable(model, z[t = 1:T])
    @variable(model, α_op >= α_op_lb)
    @variable(model, α_x >= α_x_lb)
    @constraint(model, box_lo[t = 1:T], z[t] >= 0)
    @constraint(model, box_hi[t = 1:T], z[t] <= y_inv)
    @objective(model, Min, c_y * y_inv + α_op + α_x)
    return BendersMaster(model, y_inv, z, α_op, α_x, T, Float64(c_y), Any[])
end
```

model = Model(select_optimizer(LP()))

@constraint(model, invest_op[t = 1:T], x_op[t] <= corridor_cap * x_inv)
