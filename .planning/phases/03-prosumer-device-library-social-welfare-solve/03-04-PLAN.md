  contribute!(d::PVBattery, ctx::ModelContext; T::Int) -> (; vars, p_inject, utility)
    vars     = (; p_ch, p_dch, soc)
    p_inject = [ d.Ppv[t] - p_ch[t] + p_dch[t]  for t in 1:T ]   # PV to grid − charge draw + discharge
    utility  = QuadExpr (concave charge utility − convex discharge cost)
  The device writes NOTHING to ctx.residuals and calls NO add_to_objective!.

Thesis / App. C parametrization (RESEARCH Pattern 3):
  soc[t+1] == soc[t] + (η·p_ch[t] − p_dch[t]/η)·Δt          # SOC dynamics 3.6 (round-trip η²<1)
  p_ch[t] <= Ppv[t]                                         # PV-limited charge 3.7 (charge from PV only, A6)
  0 <= p_ch[t], p_dch[t] <= Pmax                            # 3.8
  Emin <= soc[t] <= Emax ; soc[1] == soc0                   # 3.9 band + IC
  a_ch  = λ_med ; b_ch  = (λ_med − λ_min)/Pmax              # concave charge utility 3.15–3.18
  a_dch = λ_med ; b_dch = (λ_max − λ_med)/Pmax              # convex discharge cost 3.19–3.20
  utility += Σ ( a_ch·p_ch − (b_ch/2)·p_ch² − a_dch·p_dch − (b_dch/2)·p_dch² )
  NO binary, NO p_ch·p_dch==0 constraint — App. C (pp. 166–168) guarantees complementarity.
