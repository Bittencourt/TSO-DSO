function ternary_min(f, lo, hi; iters = 100)
    for _ in 1:iters
        m1 = lo + (hi - lo) / 3; m2 = hi - (hi - lo) / 3
        f(m1) < f(m2) ? (hi = m2) : (lo = m1)
    end
    z = (lo + hi) / 2
    return (z, f(z))
end

    end
