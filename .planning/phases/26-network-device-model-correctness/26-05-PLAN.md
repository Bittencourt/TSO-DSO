using TSODSO, JuMP
using TSODSO: select_optimizer, SOCP

# Standalone raw-JuMP replica of the EXACT physics :smax/:smax_rev must satisfy (rotated cone,
# true vdrop, forward + reverse apparent-power cones) -- independent of ConvexBranchFlow.jl'"'"'s
# exact internal plumbing, so this exercises the MATHEMATICAL property this task requires
# regardless of how Task 1 wires :smax_rev into contribute!.
function build_backfeed(r, x, smax; φ = 0.999999, Pmax_export = 1.0)
    tanφ = sqrt(1 - φ^2) / φ
    model = Model(select_optimizer(SOCP()))
    @variable(model, v[1:2])
    @variable(model, P)
    @variable(model, Q)
    @variable(model, l >= 0)
    @variable(model, 0 <= pexp <= Pmax_export)
    fix(v[1], 1.0; force = true)
    set_lower_bound(v[2], 0.90^2); set_upper_bound(v[2], 1.05^2)
    @constraint(model, cone, [0.5 * l, v[1], P, Q] in RotatedSecondOrderCone())
    @constraint(model, vdrop, v[2] == v[1] - 2 * (r * P + x * Q) + (r^2 + x^2) * l)
    @constraint(model, P == -pexp)
    @constraint(model, Q == -pexp * tanφ)
    @constraint(model, smax_fwd, [smax, P, Q] in SecondOrderCone())
    @constraint(model, smax_rev, [smax, P - r * l, Q - x * l] in SecondOrderCone())
    @objective(model, Max, pexp)   # elastic PV/Aggregator wants to export as much as feasible
    optimize!(model)
    return model, smax_fwd, smax_rev
end
